function config = registerpil
%

%   Copyright 2025 The MathWorks, Inc.

config(1) = rtw.connectivity.ConfigRegistry;
config(1).ConfigName = 'Arduino Board';
config(1).ConfigClass = 'codertarget.arduinotarget.pil.ConnectivityConfig';
config(1).isConfigSetCompatibleFcn = @i_isConfigSetCompatible;
end

function isConfigSetCompatible = i_isConfigSetCompatible(configSet)
% If any updates are being made to this function, then make sure that
% all the fields are checked for its existence and then used. g2258298 
% has a dot indexing error and we suspect it is due to partially loaded 
% configset data when it comes to this function. When a hardware is
% selected 'initializing board board parameter values' takes some time
% and probably this function is called before the parameters are fully
% loaded. You can see this behavior when you open a model and try 
% selecting a hardware, a pop up opens with 'Initializing board board 
% parameter values' and takes few seconds to load the board.

isConfigSetCompatible = false;
if configSet.isValidParam('CoderTargetData')
    coderTargetData =  getParam(configSet,'CoderTargetData');
    if ~isempty(coderTargetData) && isfield(coderTargetData, 'TargetHardware')
		selectedBoard = codertarget.arduinobase.board.GetBoard('boardName',coderTargetData.TargetHardware);
		if ~isempty(selectedBoard) && isprop(selectedBoard, 'isArduinoBoard')...
     		&& isprop(selectedBoard, 'boardInfo')
			if selectedBoard.isArduinoBoard
				boardInfo = selectedBoard.boardInfo;
				if isprop(boardInfo, 'IsPilSupported')
					isConfigSetCompatible = boardInfo.IsPilSupported;
				end
			end
		end
    end
end
end
